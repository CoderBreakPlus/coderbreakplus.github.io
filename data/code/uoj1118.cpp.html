inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}

std::vector<int> recover(int c,int n){
	N=n;
	all.clear();
	all.resize(n,0);
	lst=all;

	vector<int>res,A(n,(n+1)/2),B(n,(n+1)/2);
	for(int i=0;i<n;i++)res.pb(i);

	for(int l=1,r=n,tar=n-1;l<r;l++,r--,tar--){
		shuffle(res.begin(),res.end(),rnd);
		// cout<<"##at "<<l<<" "<<r<<" need tar < "<<tar<<endl;
		while(1){
			vector<int>love;
			for(int i=0;i<res.size();i++)
				if(rnd()&1)love.pb(res[i]);
			int now=qry(love);
			// cout<<"now="<<now<<endl;
			if(now<tar) break;
		}

		// cout<<"#satisfied"<<endl;

		int X1=0,X2=0;
		if(l==1){
			if(l==1){
				vector<int>V1,V2;
				while(1){
					V1.clear(),V2.clear();
					for(int x:res){
						if(rnd()&1)V1.pb(x);
						else V2.pb(x);
					}
					if(qry(V1,1)==tar)break;
				}
				auto findone=[&](vector<int>v){
					while(v.size()>1){
						int mid=v.size()/2;
						vector<int>now(v.begin(),v.begin()+mid);
						if(qry(now,1)==tar)v.resize(mid);
						else v.erase(v.begin(),v.begin()+mid);
					}
					return v[0];
				};
				X1=findone(V1);
				X2=findone(V2);
			}
		}else{
			int L=0,R=(int)res.size()-1;
			while(L<=R){
				int M=(L+R)>>1;
				vector<int>tmp;
				for(int i=0;i<=M;i++)tmp.pb(res[i]);
				int now=qry(tmp,1);
				if(now==tar){
					X1=M;
					R=M-1;
				}else L=M+1;
			}
			L=X1+1,R=(int)res.size()-1;
			while(L<=R){
				int M=(L+R)>>1;
				vector<int>tmp;
				for(int i=X1+1;i<=M;i++)tmp.pb(res[i]);
				int now=qry(tmp,1);
				if(now==tar){
					X2=M;
					R=M-1;
				}else L=M+1;
			}
			X1=res[X1],X2=res[X2];
		}
		A[X1]=A[X2]=l;
		B[X1]=B[X2]=r;
		vector<int>res2;
		for(int x:res)if(x!=X1&&x!=X2)res2.pb(x);
		swap(res,res2);
	}
	for(int i=0;i<n;i++)if(all[i])swap(A[i],B[i]);
	return A;
}
