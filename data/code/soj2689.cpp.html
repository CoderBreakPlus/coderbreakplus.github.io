inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}

namespace atcoder{
	long long floor_sum(long long n, long long m, long long a, long long b) {
	    long long ans = 0;
	    if (a >= m) {
	        ans += (n - 1) * n * (a / m) / 2;
	        a %= m;
	    }
	    if (b >= m) {
	        ans += n * (b / m);
	        b %= m;
	    }

	    long long y_max = (a * n + b) / m, x_max = (y_max * m - b);
	    if (y_max == 0) return ans;
	    ans += (n - (x_max + a - 1) / a) * y_max;
	    ans += floor_sum(y_max, a, m, (a - x_max % a) % a);
	    return ans;
	}
}

		for(int i=1;i<=n;i++)
			if(sa[i]>p) y[++w]=sa[i]-p;

inline int lcp(int a,int b){
	if(a==b)return n-a+1;
	int l=rk[a],r=rk[b];
	if(l>r)swap(l,r);
	int k=lg2(r-l++);
	return min(st[k][l],st[k][r-(1<<k)+1]);
}
namespace Border{
	int B,id[100005];
	vector<tuple<int,int,int>> v[100005];

	void init(){
		B=max(2,(int)sqrt(n));
		for(int i=1;i<=n;i++)st[0][i]=ht[i];
		for(int k=1;k<=lg2(n);k++)
			for(int i=1;i+(1<<k)-1<=n;i++)
				st[k][i]=min(st[k-1][i],st[k-1][i+(1<<(k-1))]);

		int c=0;
		for(int i=1;i<=n;i++){
			v[i].clear();
			if(i==1||ht[i]<B)c++;
			if(sa[i]+B-1<=n)id[sa[i]]=c;
		}
		for(int i=1;i+B-1<=n;i++){
			auto &a=v[id[i]];
			if(a.empty()||i-get<1>(a.back())>B/2)
				a.pb(i,i,0);
			else{
				auto &[l,r,d]=a.back();
				d=i-r;
				r=i;
			}
		}
	}
	inline void append(vector<tuple<int,int,int>> &ans,int a,int d,int c){
		if(c<=0)return;
		if(c==1)d=0;
		if(!ans.empty()){
			auto &[pa,pd,pc]=ans.back();
			int gap=a-pa-pd*(pc-1);
			if(pc==1&&(c==1||gap==d)){
				pd=gap;
				pc+=c;
				return;
			}
			if(pc>1&&gap==pd&&(c==1||d==pd)){
				pc+=c;
				return;
			}
		}
		ans.pb(a,d,c);
	}
	vector<tuple<int,int,int>> query(int l,int r){
		vector<tuple<int,int,int>> ans;
		for(int k=1;k<B&&k<r-l+1;k++)
			if(lcp(l,r-k+1)>=k)
				append(ans,k,0,1);

		if(r-l+1<=B)return ans;

		auto &a=v[id[l]];
		for(int i=(int)a.size()-1;i>=0;i--){
			auto [x,y,d]=a[i];
			if(y<=l)break;
			if(x>r-B+1)continue;

			if(!d){
				if(lcp(l,x)>=r-x+1)
					append(ans,r-x+1,0,1);
				continue;
			}

			int L=max(x,l+1),R=min(y,r-B+1);
			int len=d+lcp(l,l+d);
			int ed=x+d+lcp(x,x+d)-1;

			if(r<=ed){
				L=max(L,r-len+1);
			}else{
				int j=ed-len+1;
				L=max(L,j);
				R=min(R,j);
			}

			L=x+(L-x+d-1)/d*d;
			if(L>R)continue;
			R=x+(R-x)/d*d;

			if(lcp(l,L)>=r-L+1)
				append(ans,r-R+1,d,(R-L)/d+1);
		}
		return ans;
	}
}
