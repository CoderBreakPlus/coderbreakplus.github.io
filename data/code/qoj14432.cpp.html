inline ll qpow(ll a,ll b){
    ll ans=1, base=a;
    while(b){
        if(b&1) ans=ans*base%mod;
        base=base*base%mod; b>>=1;
    }
    return ans;
}

int n,m;
vector<int>E[300005];
int ind[300005],a[300005],vis[300005];

void procedure(){
    n=read(),m=read();

    for(int i=1;i<=m;i++){
        int u=read(),v=read();
        E[u].pb(v); ind[v]++;
    }
    queue<int>q;
    int bas=0;
    for(int i=1;i<=n;i++)
        if(!ind[i])vis[i]=1,q.push(i),bas++;

    while(!q.empty()){
        int x=q.front();q.pop();
        for(int y:E[x]){
            ind[y]--;
            if(!ind[y]) q.push(y),vis[y]=1,bas++;
        }
    }

    for(int i=1;i<=n;i++)a[i]=i;

    shuffle(a+1,a+n+1,rnd);
    int mx=0;
    for(int i=1;i<=n;i++){
        int X=a[i];if(vis[X])continue;
        int cur=1;
        queue<int>q; q.push(X),vis[X]=1;
        vector<int>node;
        while(!q.empty()){
            int x=q.front();q.pop();
            for(int y:E[x]){
                ind[y]--;
                node.pb(y);
                if(!ind[y]&&y!=X){
                    q.push(y),vis[y]=1,cur++;
                }
            }
        }
        mx=max(mx,cur);
        for(int x:node) ind[x]++;
    }

}
