void check(int x,vector<int>cur){
	int s=0, sz=cur.size();
	while(s<sz){
		int l=s,r=sz-1,ans=sz;
		while(l<=r){
			int mid=(l+r)>>1;
			vector<int>ok={x};
			for(int i=l;i<=mid;i++) ok.pb(cur[i]);
			if(qry(ok)){
				ans=mid;
				r=mid-1;
			}else l=mid+1;
		}
		// if(ans==sz)break;
		link(x,cur[ans]);
		s=ans+1;

		vector<int>tmp={x};
		for(int i=ans+1;i<sz;i++)
			tmp.pb(cur[i]);
		if(!qry(tmp)) break;
	}
}
void solve(vector<int>v){
	if(v.empty())return;
	int sz=v.size();
	vector<int>cur={v[0]},icur;
	for(int i=1;i<sz;i++){
		cur.pb(v[i]);
		if(qry(cur)) cur.pop_back(),icur.pb(v[i]);
	}
	for(int x:icur) check(x,cur);
	solve(icur);
}
