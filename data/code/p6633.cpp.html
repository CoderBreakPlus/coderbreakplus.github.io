inline void NTT(vector<int> &a, int len, int g) {
	vector<int> rev(len);
	vector<ull> f(len);
	for (int i = 0; i < len; ++i) {
		rev[i] = (rev[i >> 1] >> 1) | ((i & 1) ? (len >> 1) : 0);
		f[i] = a[rev[i]];
	}
	for (int i = 1; i < len; i <<= 1) {
		int wn = qpow(g, (mod - 1) / (i << 1));
		vector<int> w(i);
		w[0] = 1;
		for (int j = 1; j < i; ++j) {
			w[j] = (ull)w[j - 1] * wn % mod;
		}
		for (int j = 0; j < len; j += (i << 1)) {
			for (int k = 0; k < i; ++k) {
				ull s = f[j + k], t = f[i + j + k] * w[k] % mod;
				f[j + k] = s + t;
				f[i + j + k] = s - t + mod;
			}
		}
		if (i == (1 << 16)) {
			for (int j = 0; j < len; ++j) {
				f[j] %= mod;
			}
		}
	}
	for (int j = 0; j < len; ++j) {
		a[j] = f[j] % mod;
	}
}

inline poly poly::inv() const {

	while (len < (int)size()) {
		len <<= 1;
		f.resize(len << 1), res.resize(len << 1);
		for (int i = 0; i < len; ++i) {
			if (i >= (int)size()) {
				break;
			}
			f[i] = data[i];
		}
		NTT(f, len << 1, G);
		NTT(res, len << 1, G);
		for (int i = 0; i < (len << 1); ++i) {
			int t = (ull)f[i] * res[i] % mod * res[i] % mod;
			addmod(res[i] <<= 1);
			addmod(res[i] += mod - t);
		}
		NTT(res, len << 1, GI);
		int ilen = qinv(len << 1);
		for (int i = 0; i < len; ++i) {
			res[i] = (ull)res[i] * ilen % mod;
		}
		for (int i = len; i < (len << 1); ++i) {

		}
