inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}

int n,m,q,l[200005],c[200005];

struct BIT{
	ll c[200005];
	void upd(ll x,ll w){
		while(x<=m+1){
			c[x]+=w;
			x+=(x&-x);
		}
	}
	ll qry(int x){
		ll ret=0;
		while(x){
			ret+=c[x];
			x-=(x&-x);
		}
		return ret;
	}
}B0,B1;
