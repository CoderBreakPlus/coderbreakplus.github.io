inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}

int n,m,q;

struct BIT{
	ll c[1000005];
	void upd(int x,ll w){
		// cout<<"upd "<<x<<" with "<<w<<endl;
		while(x<=n){
			c[x]+=w;
			x+=(x&-x);
		}
	}
	ll qry(int x){
		ll ret=0;
		while(x){
			ret+=c[x];
			x-=(x&-x);
		}
		return ret;
	}
}B;

void procedure(){
	n=read(),m=read();	
	for(int i=m;i<=n;i++)s[i]=read()+s[i-m];
	for(int i=n;i>=1;i--)s[i]-=s[i-1];

	q=read();
	for(int i=1;i<=q;i++){
		l[i]=read(),r[i]=read();

	// for(int i=1;i<=n;i++) cout<<s[i]<<" "; cout<<endl;
	for(int i=1;i<=n;i++){
		// cout<<"r="<<i<<endl;

		if(i>m){
			int lst=stk[o].back();
			B.upd(lst,-s[i]);
			while(!stk[o].empty()&&s[lst]<s[i]){
				ll dc=s[lst]-s[i];
				B.upd(lst,-dc);
				stk[o].pop_back();
				if(!stk[o].empty()) B.upd(lst=stk[o].back(),dc);
