inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}
inline ll INV(ll x){ return qpow(x, mod-2); }

int m;
ll a[200005],tmp[200005];
struct BIT{
	ll c[200005];
	void upd(ll x,ll w){
		while(x<=m){
			c[x]+=w;
			x+=(x&-x);
		}
	}
	ll qry(ll x){
		ll ret=0;
		while(x){
			ret+=c[x];
			x-=(x&-x);
		}
		return ret;
	}
	ll findkth(ll k){
		ll l=1,r=m,ans=0;
		while(l<=r){
			ll mid=(l+r)>>1;
			if(qry(mid)>=k) ans=mid,r=mid-1;
			else l=mid+1;
		}
		return ans;
	}
}b1,b2;
