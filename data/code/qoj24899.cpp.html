inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}

int prv0[300005],nxt0[300005];
int prv1[300005],nxt1[300005];

int stk[300005],tp;

ll suml[300005],sumr[300005],ans[300005];

void procedure(){
	n=read();
	for(int i=1;i<=n;i++)h[i]=read();
	for(int i=1;i<=n;i++)w[i]=read();

	for(int i=1;i<=n;i++)prv0[i]=prv1[i]=0;
	for(int i=1;i<=n;i++)nxt0[i]=nxt1[i]=n+1;

	for(int i=1;i<=n;i++){
		while(tp&&h[stk[tp]]<h[i])
			nxt1[stk[tp--]]=i;
		if(tp)prv0[i]=stk[tp];
		stk[++tp]=i;
	}
	tp=0;
	for(int i=1;i<=n;i++){
		while(tp&&h[stk[tp]]<=h[i])
			nxt0[stk[tp--]]=i;
		if(tp)prv1[i]=stk[tp];
		stk[++tp]=i;

	for(int i=1;i<=n;i++)
		suml[i]=suml[prv1[i]]+w[i];
	for(int i=n;i>=1;i--)
		sumr[i]=sumr[nxt1[i]]+w[i];

	for(int i=1;i<=n;i++){

			ans[i+1]-=val;
			ans[nxt0[i]]+=val;
