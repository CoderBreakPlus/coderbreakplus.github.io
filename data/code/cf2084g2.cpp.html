inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}

mt19937 rnd(chrono::steady_clock::now().time_since_epoch().count());
int n,a[500005];

int lc[500005],rc[500005],k[500005],sz[500005],rt;
ll t1[500005],t2[500005],val[500005],v0,v1;

int cnt;

int newnode(){
	int x=++cnt;
	lc[x]=rc[x]=t1[x]=t2[x]=val[x]=0,k[x]=rnd(),sz[x]=1;
	return x;
}
void upd(int x,ll s1,ll s2){
	if(x){
		t1[x]+=s1, t2[x]+=s2;
		val[x]+=s1+sz[lc[x]]*s2;
	}
}
void pushdown(int x){
	upd(lc[x],t1[x],t2[x]);
	upd(rc[x],t1[x]+t2[x]*(sz[lc[x]]+1),t2[x]);
	t1[x]=t2[x]=0;
}
void pushup(int x){
	sz[x]=sz[lc[x]]+sz[rc[x]]+1;
}
int merge(int x,int y){
	if(!x || !y) return x^y;
	pushdown(x),pushdown(y);
	if(k[x]<k[y]){
		rc[x]=merge(rc[x],y);
		return pushup(x),x;
	}else{
		lc[y]=merge(x,lc[y]);
		return pushup(y),y;
	}
}
void split(int x,int &a,int &b){
	if(!x){ a=b=0; return; }
	pushdown(x);
	if(val[x]>=0) {
		a=x;
		split(rc[x],rc[x],b);
	}else{
		b=x;
		split(lc[x],a,lc[x]);
	}
	pushup(x);
}
vector<ll>vec;
void divide(int x){
	if(!x)return;
	pushdown(x);
	divide(rc[x]);
	vec.pb(val[x]);
	divide(lc[x]);
}
