const int mod = 998244353;
inline ll read(){
	ll x=0, f=1; int ch=getchar();
	while(ch<'0' || ch>'9') { if(ch=='-') f=-1; ch=getchar(); }
	while(ch>='0' && ch<='9') x=x*10+ch-'0', ch=getchar();
	return x*f;
}
inline int lg2(int x){ return 31^__builtin_clz(x); }
inline ll lg2(ll x){ return 63^__builtin_clzll(x); }
inline void addmod(int &x){ if(x >= mod) x -= mod; }
inline void addmod(ll &x){ if(x >= mod) x -= mod; }
inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}
inline ll INV(ll x){ return qpow(x, mod-2); }

int n,a[100005],b[100005],ind[100005],ok[100005];
int c[100005],cc,d[100005],dd;
vector<int>E[100005];
vector<pair<int,int>>edge;

int stk[100005],tp;
void add(int u,int v){
	// cout<<u<<"->"<<v<<endl;
	E[u].pb(v); edge.pb(u,v); ind[v]++;
}
void procedure(){
	n=read();
	for(int i=1;i<=n;i++) a[i]=read();
	for(int i=1;i<=n;i++) b[i]=read();

	for(int i=1;i<=n;i++){
		while(tp && a[stk[tp]]<a[i]) tp--;
		if(tp) add(a[stk[tp]],a[i]);
		stk[++tp]=i;
	}
	tp=0;
	for(int i=1;i<=n;i++){
		while(tp && b[stk[tp]]>b[i]) tp--;
		if(tp) add(b[stk[tp]],b[i]);
		stk[++tp]=i;
	}

	{
		priority_queue<int,vector<int>,greater<int>>q;
		for(int i=1;i<=n;i++)if(!(ok[i]=ind[i]))q.push(i);
		while(!q.empty()){
			int x=q.top();q.pop();
			c[++cc]=x;
			for(auto y:E[x]){
				ok[y]--;
				if(!ok[y]) q.push(y);
			}
		}
		// for(int i=1;i<=n;i++) cout<<c[i]<<" "; cout<<endl;
		for(int i=1;i<=n;i++)if(a[i]!=c[i]){puts("No");return;}
	}
	{
		priority_queue<int>q;
		for(int i=1;i<=n;i++)if(!(ok[i]=ind[i]))q.push(i);
		while(!q.empty()){
			int x=q.top();q.pop();
			d[++dd]=x;
			for(auto y:E[x]){
				ok[y]--;
				if(!ok[y]) q.push(y);
			}
		}
		for(int i=1;i<=n;i++)if(b[i]!=d[i]){puts("No");return;}
	}
	puts("Yes");
	printf("%d\n",(int)edge.size());
	for(auto [x,y]: edge){
		printf("%d %d\n",x,y);
	}
}
