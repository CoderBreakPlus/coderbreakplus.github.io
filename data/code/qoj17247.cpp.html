inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}

void procedure(){
	n=read(),m=read();

	for(int i=0;i<=n;i++){
		C[i][0]=1;
		for(int j=1;j<=i;j++)
			addmod(C[i][j]=C[i-1][j]+C[i-1][j-1]);
	}
	for(int i=1;i<=m;i++){
		int x=read(),y=read();
		pos[x]=y;
	}
	g[0]=1;
	for(int i=1;i<=n;i++)
		for(int j=0;j<i;j++) add(g[i],1ull*g[j]*g[i-j-1]%mod);

	for(int x=1;x<=n;x++){
		int mn=1e9, mx=-1e9;
		for(int y=x;y<=n;y++){
			if(pos[y]){
				mn=min(mn,pos[y]);
				mx=max(mx,pos[y]);
			}
			l[x][y]=mn, r[x][y]=mx;
		}
	}

	for(int x=1;x<=n+1;x++)
		for(int z=0;z<=n+1;z++) f[x][x-1][z]=1;
	for(int x=n;x>=1;x--)
		for(int y=x;y<=n;y++){
			int len=y-x+1;
			int L=max(1,r[x][y]-len+1),R=min(n-len+1,l[x][y]);

			memset(tmp,0,sizeof(tmp));

			if(l[x][y]>r[x][y]){
				for(int z=L;z<=R;z++) f[x][y][z]=g[len];

							tmp[z]+=1ull*F[z]*G[z+t+1];
