inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}

namespace SAM{
	int len[M<<1],fa[M<<1],son[M<<1][4],lst;
	inline void ins(int p,int v){
		len[p]=len[lst]+1;
		while(lst&&!son[lst][v]){
			son[lst][v]=p;
			lst=fa[lst];
		}
		if(!son[lst][v]){
			son[lst][v]=p;
		}else{
			int q=son[lst][v];
			if(len[q]==len[lst]+1){
				fa[p]=q;
			}else{
				int u=++m;
				len[u]=len[lst]+1,fa[u]=fa[q];
				memcpy(son[u],son[q],sizeof(son[u]));
				fa[p]=fa[q]=u;
				while(son[lst][v]==q){
					son[lst][v]=u;
					lst=fa[lst];
				}
			}
		}
		lst=p;
	}
	void clear(){
		for(int i=0;i<=m;i++)len[i]=fa[i]=0,memset(son[i],0,sizeof(son[i]));
		lst=0;
	}
}

bool dfs(int x){
	int L=tim+1;
	if(1<=x&&x<=n)seq[++tim]=x;
	bool flg=0;
	for(int y:E[x])flg|=dfs(y);
	int R=tim;
	if(!flg&&vis[x]){
		// cout<<"limit length="<<vis[x]<<endl;
		sort(seq+L,seq+R+1);
		int lst=1;
		for(int i=L;i<=R;i++){
			chkmax(ans[lst],seq[i]);
			lst=seq[i]-vis[x]+2;
		}
		chkmax(ans[lst],n+1);
		return 1;
	}
	return flg;
}
