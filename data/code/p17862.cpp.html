struct Barrett {
	using u32 = uint32_t;
	using u64 = uint64_t;
	using u128 = __uint128_t;
	u32 mod; u64 inv;
	explicit Barrett(u32 p) : mod(p), inv(UINT64_MAX / p) {}
	inline u32 reduce(u64 x) const {
		u64 q = (u128)x * inv >> 64;
		u64 r = x - q * mod;
		return r >= mod ? r - mod : r;
	}
	inline u32 mul(u32 a, u32 b) const {
		return reduce((u64)a * b);
	}
