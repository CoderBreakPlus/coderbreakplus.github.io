inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}

void dfs(int x,int fa){
	for(auto y:E[x]){
		if(y==fa)continue;
		dfs(y,x);
	}

int que[100005],hd,tl;

void merge(int x,int y){
	x=find(x),y=find(y);
	if(x!=y)
	sz[x]+=sz[y],fa[y]=x;
}
void procedure(){
	n=read(),m=read(),q=read();
	for(int i=1;i<=n;i++)fa[i]=i,sz[i]=1;
	for(int i=1;i<=m;i++){
		int u=read(),v=read();
		E[u].pb(v),E[v].pb(u);
		merge(u,v);
	}
	for(int i=1;i<=n;i++) rt[i]=i;
	shuffle(rt+1,rt+n+1,rnd);
	int K=min(200,n);

	for(int i=1;i<=K;i++){
		que[hd=tl=1]=rt[i];
		dis[i][rt[i]]=1;

		while(hd<=tl){
			int x=que[hd++];
			for(auto y:E[x]){
				if(dis[i][y])continue;
				dis[i][y]=dis[i][x]+1;
				que[++tl]=y;
			}
		}
	}
	while(q--){
		int u=read(),v=read(),ok=0;

			puts("YES");

		for(int i=1;i<=K;i++)
			if(dis[i][u]&&dis[i][v])
			ok|=(dis[i][u]+dis[i][v]-2<=20000);
		puts(ok?"NO":"YES");
	}
}
