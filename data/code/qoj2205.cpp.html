inline ll read(){
	ll x=0, f=1; int ch=getchar();

	while(ch>='0' && ch<='9') x=x*10+ch-'0', ch=getchar();
	return x*f;
}
inline int lg2(int x){ return 31^__builtin_clz(x); }
inline ll lg2(ll x){ return 63^__builtin_clzll(x); }
const ll mod = (1ll<<50);
template<typename T>inline void addmod(T &x){ if(x >= mod) x -= mod; }

#define u128 __uint128_t
inline ll qpow(ll a,ll b,ll m){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=(u128)ans*base%m;
		base=(u128)base*base%m; b>>=1;
	}
	return ans;
}

ll n,a[200005],b[200005],c[400005],fac[200005],inv[200005];

ll pool[10000005]; ll *cur=pool;
void fastmul(ll *a,ll *b,ll *c,ll n){
	if(n<=32){
		for(int i=0;i<=2*n;i++)c[i]=0;
		for(int i=0;i<=n;i++)
			for(int j=0;j<=n;j++)
				c[i+j]=(c[i+j]+(u128)a[i]*b[j])%mod;
		return;
	}
	ll sz1=n>>1, sz2=n-sz1-1;
	ll *cur0=cur;
	ll *c1=cur; cur+=2*sz1+1;
	fastmul(a,b,c1,sz1);

	ll *c2=cur; cur+=2*sz2+1;
	fastmul(a+sz1+1,b+sz1+1,c2,sz2);

	ll *a3=cur; cur+=sz1+1;
	ll *b3=cur; cur+=sz1+1;
	ll *c3=cur; cur+=2*sz1+1;
	for(ll i=0;i<=sz1;i++){
		addmod(a3[i]=a[i]+(i+sz1+1<=n?a[i+sz1+1]:0));
		addmod(b3[i]=b[i]+(i+sz1+1<=n?b[i+sz1+1]:0));
	}
	fastmul(a3,b3,c3,sz1);

	for(int i=0;i<=2*sz1;i++) c3[i]=c3[i]+2*mod-c1[i]-(i<=2*sz2?c2[i]:0);
	for(int i=0;i<=2*n;i++) c[i]=0;

	for(int i=0;i<=2*sz1;i++) c[i]+=c1[i];
	for(int i=0;i<=2*sz2;i++) c[i+2*(sz1+1)]+=c2[i];
	for(int i=0;i<=sz1+sz2;i++) c[i+sz1+1]+=c3[i];

	for(int i=0;i<=2*n;i++) c[i]%=mod;
	cur=cur0;
}
