inline ll qpow(ll a,ll b){

pair<ll,ll> cake(ll N){
	n=N,m=0,sum=0;

	ll L=0,R=0;
	for(ll k=1;;){
		int cnt=0;
		bool same=0;
		for(int T=0;T<6;T++){
			char c=add(k);
			if(c=='='){
				m=k;
				same=1;
				break;
			}
			if(c=='<')cnt++;
		}
		if(same)break;

		if(cnt>=4){
			L=k+1;
			R=min(n,2*k-1);
			break;
		}

		if(k==n){
			m=n;
			break;
		}
		ll nk=(ll)min<__int128>(n,((__int128)3*k+1)/2);
		if(nk<=k)nk=k+1;
		k=nk;
	}
	
	if(!m){
		const int T=8;
		while(L<R){
			ll M=(L+R)>>1;
			int cnt=0;
			bool same=0;
			for(int t=0;t<T;t++){
				char c=add(M);
				if(c=='='){
					m=M;
					same=1;
					break;
				}
				if(c=='<')cnt++;
			}
			if(same)break;

			ll lim=(ll)(((__int128)T*(M-L)+L-1)/L);

			if(cnt<=lim)R=M-1;
			else L=M+1;
		}
		if(!m)m=L;
	}

	if(m==1)return {0,1};

	ll s=(ll)(sum%m);
	if(s)add(m-s);

	L=0,R=m-1;
	while(L<R){
		ll M=(L+R+1)>>1;
		char c=add(m-M);
		add(M);
		if(c=='<')L=M;
		else R=M-1;
	}
	return {L,m};
}
