inline ll read(){

	while(ch>='0' && ch<='9') x=x*10+ch-'0', ch=getchar();
	return x*f;
}

inline ll qpow(ll a,ll b){
	ll ans=1, base=a;
	while(b){
		if(b&1) ans=ans*base%mod;
		base=base*base%mod; b>>=1;
	}
	return ans;
}
inline ll INV(ll x){ return qpow(x, mod-2); }

int n;

void procedure(){
	vector<pair<int,int>>v1,v2;

	n=read();
	scanf("%s%s",a+1,b+1);

	int cnt=0, L=0, R=0, sb=0;
	for(int i=1;i<=n;i++){
		cp[i]=a[i],cq[i]=b[i];

	}

	if(!L){
		puts("Yes\n0");
		return;

	reverse(a+L+1,a+R);

	int N=R-L-1;

	auto push1 = [&](int x,int y){
		x-=L,y-=L;
		v1.pb(N-y+1+L,N-x+1+L);
	};
	auto push2 = [&](int x,int y){
		v2.pb(x,y);
	};

	// B...A -> A... B on rev(X) and Y

		}

		for(int i=0;i<sa.size();i++){
			if(sa[i]==sb[i]) continue;
			if(sa[i]<sb[i]) push2(sa[i],sb[i]);
			else push1(sb[i],sa[i]);

			if(b[i]=='B')sb.pb(i);
		}
		for(int i=0;i<sa.size();i++){
			if(sa[i]==sb[i]) continue;
			if(sa[i]>sb[i]) push2(sb[i],sa[i]);
			else push1(sa[i],sb[i]);
		}
